import { useState } from 'react';
import type { Evidence } from './TestEvidence';
import { githubLine, SourceLines } from './SourcePane';
export interface AssertionSource { path: string; label: string; source: string; sha256: string; startLine: number; endLine: number; assertLines: number[] }
export function AssertionsPane({ evidence }: { evidence: Evidence }) {
  const [index, setIndex] = useState(() => Math.max(0, evidence.assertions.findIndex((source) => source.assertLines.length > 0)));
  const snippet = evidence.assertions[index]!;
  return <section className="pane assertions-pane" aria-label="Test assertions"><header className="pane-heading"><span>04 / TEST CONTRACT</span><h2>Test assertions</h2></header>
    <p>{evidence.asserts || 'Literal named test and referenced local helpers from this checkout.'}</p>
    <p className="scope">The selected test passed. Blue marks assertion call sites, not statement coverage. Helpers are linked statically; individual branches and external assertion methods are not claimed as traced.</p>
    <label>Test source <select aria-label="Test source" value={index} onChange={(event) => setIndex(Number(event.target.value))}>{evidence.assertions.map((source, position) => <option key={`${source.path}:${source.startLine}`} value={position}>{source.label} · {source.path}:{source.startLine}</option>)}</select></label>
    <a href={githubLine(evidence.sha, snippet.path, snippet.startLine)}>Pinned test assertions on GitHub ↗</a><p className="hash">Test file SHA256: {snippet.sha256}</p>
    <SourceLines source={snippet.source} start={snippet.startLine} path={snippet.path} sha={evidence.sha} assertLines={snippet.assertLines} focusLine={Math.max(snippet.startLine, (snippet.assertLines[0] ?? snippet.startLine) - 3)} />
  </section>;
}
