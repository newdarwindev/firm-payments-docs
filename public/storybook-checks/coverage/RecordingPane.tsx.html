import { useState, type ReactNode } from 'react';
import type { Evidence } from './TestEvidence';
export function RecordingFrame({ children }: { children: ReactNode }) {
  return <section className="pane recording-pane" aria-label="Terminal recording"><header className="pane-heading"><span>02 / RECORDED DEMO</span><h2>Actual test execution</h2></header>{children}</section>;
}
export function RecordingBody({ evidence, mediaBase }: { evidence: Evidence; mediaBase: string }) {
  const [play, setPlay] = useState(false);
  const asset = (path: string) => `${mediaBase}${path}?revision=${evidence.sha}`;
  return <><h3>{evidence.name}</h3><p className="summary" role="status">{evidence.status.toUpperCase()} · {evidence.total ? `${evidence.covered} / ${evidence.total} Go statements covered (${(100 * evidence.covered / evidence.total).toFixed(2)}%)` : 'No instrumented Go statements in this test package'}</p>
    <p className="scope">{evidence.scope}</p><button onClick={() => setPlay(!play)}>{play ? 'Pause terminal recording' : 'Play terminal recording'}</button>
    <img className="recording" loading="lazy" width={1440} height={900} alt={`Terminal execution of ${evidence.name}`} src={asset(play ? 'terminal.gif' : 'poster.png')} />
    <nav aria-label="Downloads"><a href={asset('terminal.gif')}>Direct terminal GIF</a><a href={asset('terminal.txt')}>Terminal text</a><a href={asset('coverage.out')}>Raw coverage profile</a><a href={asset('coverage.html')}>Go coverage HTML</a><a href={asset('data.json')}>Evidence JSON</a></nav>
    <details><summary>Executed command and test selection</summary><pre>{evidence.command.join(' ')}</pre><p>Prepared once for this package:</p><pre>{evidence.build.join(' ')}</pre><p className="hash">Test executable SHA256: {evidence.binarySHA256}</p><ul>{evidence.executed.map((name) => <li key={name}>{name}</li>)}</ul></details></>;
}
export function RecordingPane({ evidence, mediaBase, children }: { evidence: Evidence; mediaBase: string; children?: ReactNode }) {
  return <RecordingFrame>{children}<RecordingBody key={evidence.id} evidence={evidence} mediaBase={mediaBase} /></RecordingFrame>;
}
