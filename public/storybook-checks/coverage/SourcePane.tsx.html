import { useEffect, useRef, useState } from 'react';
import type { Block, Evidence } from './TestEvidence';
export const githubLine = (sha: string, path: string, line: number) => `https://github.com/newdarwindev/firm-payments/blob/${sha}/${path}#L${line}`;
export function lineState(blocks: Block[], line: number) {
  const active = blocks.filter((block) => block.statements > 0 && block.startLine <= line && block.endLine >= line && !(block.endLine === line && block.endColumn === 1));
  if (!active.length) return 'neutral';
  if (active.every((block) => block.count > 0)) return 'covered';
  return active.some((block) => block.count > 0) ? 'mixed' : 'uncovered';
}
export function SourceLines({ source, start = 1, sha, path, blocks = [], assertLines = [], focusLine }: { focusLine?: number; source: string; start?: number; sha: string; path: string; blocks?: Block[]; assertLines?: number[] }) {
  const container = useRef<HTMLDivElement>(null);
  useEffect(() => { const line = container.current?.querySelector<HTMLElement>(`[data-line="${focusLine}"]`); if (line && container.current) container.current.scrollTop = line.offsetTop; }, [source, focusLine]);
  return <div ref={container} className="code" tabIndex={0} aria-label={`Source code: ${path}`}><pre>{source.split('\n').map((text, index) => {
    const line = start + index;
    const state = assertLines.includes(line) ? 'assertion' : lineState(blocks, line);
    return <span className={`line ${state}`} data-line={line} key={line} title={state === 'assertion' ? 'Assertion call site; execution of this branch is not inferred' : `Statement coverage: ${state}`}><a href={githubLine(sha, path, line)} target="_blank" rel="noreferrer" aria-label={`GitHub line ${line}`}>{line}</a><code>{text || ' '}</code></span>;
  })}</pre></div>;
}
export function CodePane({ evidence, initialFile, initialLine, onFileChange }: { evidence: Evidence; initialFile?: string; initialLine?: number; onFileChange?: (path: string) => void }) {
  const [path, setPath] = useState(evidence.files.find((file) => file.path === initialFile)?.path ?? evidence.files[0]?.path ?? '');
  const file = evidence.files.find((candidate) => candidate.path === path);
  return <section className="pane code-pane" aria-label="Covered implementation"><header className="pane-heading"><span>03 / IMPLEMENTATION</span><h2>Covered code</h2></header>
    <p className="legend">Green: covered · Red: uncovered · Amber: mixed blocks · Gray: uninstrumented. Line numbers open this revision on GitHub.</p>
    {file ? <><label>Source file <select aria-label="Source file" value={path} onChange={(event) => { setPath(event.target.value); onFileChange?.(event.target.value); }}>{evidence.files.map((candidate) => <option key={candidate.path} value={candidate.path}>{candidate.path}</option>)}</select></label>
      <a href={githubLine(evidence.sha, file.path, 1)}>Pinned implementation on GitHub ↗</a><p className="hash">Source SHA256: {file.sha256}</p>
      <SourceLines source={file.source} sha={evidence.sha} path={file.path} blocks={file.blocks} focusLine={path === initialFile && initialLine ? initialLine : file.blocks.find((block) => block.count > 0)?.startLine} />
      <details><summary>Exact statement blocks and execution counts</summary><table><thead><tr><th>Position</th><th>Statements</th><th>Count</th></tr></thead><tbody>{file.blocks.map((block) => <tr key={`${block.startLine}:${block.startColumn}`}><td>{block.startLine}:{block.startColumn}–{block.endLine}:{block.endColumn}</td><td>{block.statements}</td><td>{block.count}</td></tr>)}</tbody></table></details>
    </> : <p>No instrumented Go statements in this test package.</p>}
  </section>;
}
