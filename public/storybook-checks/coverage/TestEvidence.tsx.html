import { useState } from 'react';
import './style.css';
 
export interface Block { startLine: number; startColumn: number; endLine: number; endColumn: number; statements: number; count: number }
export interface FileCoverage { path: string; sha256: string; source: string; blocks: Block[] }
export interface Evidence {
  id: string; suite: string; name: string; sha: string; status: string; asserts: string; scope: string;
  selected: string; command: string[]; build: string[]; binarySHA256: string; executed: string[]; group: boolean; covered: number; total: number; files: FileCoverage[];
}
export function lineState(blocks: Block[], line: number) {
  const active = blocks.filter((block) => block.statements > 0 && block.startLine <= line && block.endLine >= line && !(block.endLine === line && block.endColumn === 1));
  if (!active.length) return 'neutral';
  if (active.every((block) => block.count > 0)) return 'covered';
  return active.some((block) => block.count > 0) ? 'mixed' : 'uncovered';
}
export function TestEvidence({ evidence, mediaBase, initialFile, onFileChange }: { evidence: Evidence; mediaBase: string; initialFile?: string; onFileChange?: (path: string) => void }) {
  const [path, setPath] = useState(evidence.files.find((file) => file.path === initialFile)?.path ?? evidence.files[0]?.path ?? '');
  const [play, setPlay] = useState(false);
  const file = evidence.files.find((candidate) => candidate.path === path);
  return <article className="evidence">
    <p className="eyebrow">{evidence.suite} · {evidence.group ? 'test group' : 'individual test'} · {evidence.sha.slice(0, 7)}</p>
    <h1>{evidence.name}</h1>
    <p className="summary" role="status">{evidence.status.toUpperCase()} · {evidence.total ? `${evidence.covered} / ${evidence.total} Go statements covered (${(100 * evidence.covered / evidence.total).toFixed(2)}%)` : 'No instrumented Go statements in this test package'}</p>
    <p>{evidence.asserts}</p><p className="scope">{evidence.scope}</p>
    <details><summary>Executed command and test selection</summary><p>Prepared once for this package:</p><pre>{evidence.build.join(' ')}</pre><p>Test executable SHA256: {evidence.binarySHA256}</p><pre>{evidence.command.join(' ')}</pre><ul>{evidence.executed.map((name) => <li key={name}>{name}</li>)}</ul></details>
    <nav aria-label="Downloads"><a href={`${mediaBase}coverage.out`}>Raw coverage profile</a><a href={`${mediaBase}coverage.html`}>Downloadable Go coverage HTML</a><a href={`${mediaBase}data.json`}>Evidence JSON</a><a href={`${mediaBase}terminal.gif`}>Direct terminal GIF</a><a href={`${mediaBase}terminal.txt`}>Terminal text</a></nav>
    {file && <section aria-label="Code coverage">
      <label>Source file <select aria-label="Source file" value={path} onChange={(event) => { setPath(event.target.value); onFileChange?.(event.target.value); }}>{evidence.files.map((candidate) => <option key={candidate.path} value={candidate.path}>{candidate.path}</option>)}</select></label>
      <p className="legend">Green: covered · Red: uncovered · Amber: mixed statement blocks · Gray: uninstrumented</p>
      <p className="hash">Source SHA256: {file.sha256}</p>
      <div className="code" tabIndex={0} aria-label={`Source code: ${file.path}`}><pre>{file.source.split('\n').map((text, index) => <span className={`line ${lineState(file.blocks, index + 1)}`} id={`L${index + 1}`} key={index}><a href={`#L${index + 1}`} aria-label={`Line ${index + 1}`}>{index + 1}</a><code>{text || ' '}</code></span>)}</pre></div>
      <details><summary>Exact statement blocks and execution counts</summary><table><thead><tr><th>Position</th><th>Statements</th><th>Count</th></tr></thead><tbody>{file.blocks.map((block) => <tr key={`${block.startLine}:${block.startColumn}`}><td>{block.startLine}:{block.startColumn}–{block.endLine}:{block.endColumn}</td><td>{block.statements}</td><td>{block.count}</td></tr>)}</tbody></table></details>
    </section>}
    <section aria-label="Terminal recording"><h2>Actual test execution</h2><button onClick={() => setPlay(!play)}>{play ? 'Pause terminal recording' : 'Play terminal recording'}</button><img className="recording" alt={`Terminal execution of ${evidence.name}`} src={`${mediaBase}${play ? 'terminal.gif' : 'poster.png'}`} /></section>
  </article>;
}
